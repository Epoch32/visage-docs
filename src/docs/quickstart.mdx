Get up and running with Visage in under two minutes using Bun, pnpm, or npm.

## Installation

Install Visage and the official Vite plugin for TypeScript support, fast HMR, and AOT template compilation:

<CodeBlock
  filename="terminal"
  lang="bash"
  code={`# Using Bun
bun add @epoch32/visage

# Or using pnpm
pnpm add @epoch32/visage`}
/>

## Vite Configuration

<CodeBlock
  filename="vite.config.ts"
  lang="ts"
  code={`import { defineConfig } from "vite";
import visage from "@epoch32/visage/vite";

export default defineConfig({
  plugins: [
    visage({
      devtools: true, // Auto-inject DAG devtools in development
      hmr: true,      // Fast signal-preserving HMR
    }),
  ],
});`}
/>

## Your First Component

<CodeBlock
  filename="src/main.tsx"
  lang="tsx"
  code={`import { signal, render } from "@epoch32/visage";

function App() {
  const name = signal("World");

  return (
    <div class="p-6 max-w-md mx-auto space-y-4">
      <h1 class="text-2xl font-bold">Hello, {name}!</h1>
      <input
        type="text"
        value={name}
        onInput={(e) => name.set(e.currentTarget.value)}
        class="border border-zinc-700 bg-zinc-900 px-3 py-2 rounded w-full"
      />
    </div>
  );
}

render(() => <App />, document.getElementById("app")!);`}
/>
