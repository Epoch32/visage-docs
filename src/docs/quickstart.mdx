Get up and running with Visage in under two minutes using Bun or npm.

## Installation

Install Visage and the official Vite plugin for TypeScript support and fast HMR:

<CodeBlock
  filename="terminal"
  lang="bash"
  code={`# Using Bun
bun add @epoch32/visage

# Or using pnpm
pnpm add @epoch32/visage`}
/>

## Vite Configuration

<CodeBlock
  filename="vite.config.ts"
  lang="ts"
  code={`import { defineConfig } from "vite";
import visage from "@epoch32/visage/vite";

export default defineConfig({
  plugins: [visage()]
});`}
/>

## Your First Component

<CodeBlock
  filename="src/main.tsx"
  lang="tsx"
  code={`import { createSignal, render } from "@epoch32/visage";

function App() {
  const [name, setName] = createSignal("World");

  return (
    <div>
      <h1>Hello, {name}!</h1>
      <input
        value={name()}
        onInput={(e) => setName((e.target as HTMLInputElement).value)}
      />
    </div>
  );
}

render(() => <App />, document.getElementById("app")!);`}
/>
